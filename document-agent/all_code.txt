_llm_client = None

def _get_llm_client():
    global _llm_client
    if _llm_client is not None:
        return _llm_client
    key = os.getenv("AZURE_OPENAI_KEY")
    endpoint = os.getenv("AZURE_OPENAI_ENDPOINT")
    if not key or not endpoint:
        return None
    from openai import AzureOpenAI
    _llm_client = AzureOpenAI(
        api_key=key,
        azure_endpoint=endpoint,
        api_version=os.getenv("AZURE_OPENAI_API_VERSION", "2023-07-01-preview"),
    )
    return _llm_client


def _translate_chunk(text: str) -> str:
    """Translate a single chunk of French text to English using Azure OpenAI."""
    client = _get_llm_client()
    if client is None:
        raise RuntimeError(
            "Azure OpenAI not configured. Set AZURE_OPENAI_KEY and "
            "AZURE_OPENAI_ENDPOINT in your .env file."
        )
    model = os.getenv("AZURE_OPENAI_DEPLOYMENT", "gpt-4o-mini")
    resp = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": TRANSLATION_SYSTEM_PROMPT},
            {"role": "user", "content": text},
        ],
        max_tokens=4096,
        temperature=0.1,
    )
    return resp.choices[0].message.content.strip()


def translate_document(text: str, progress_callback=None) -> str:
    """
    Translate full document text from French to English.
    Splits into manageable chunks to stay within token limits.
    """
    max_chars = 8000  # safe for gpt-4o-mini context
    if len(text) <= max_chars:
        if progress_callback:
            progress_callback(0, 1)
        result = _translate_chunk(text)
        if progress_callback:
            progress_callback(1, 1)
        return result

    # Split by page markers first, then by paragraphs
    pages = re.split(r'(--- Page \d+ ---)', text)
    chunks = []
    current = ""

    for part in pages:
        if len(current) + len(part) + 2 > max_chars and current:
            chunks.append(current)
            current = part
        else:
            current += ("\n\n" if current else "") + part
    if current:
        chunks.append(current)

    translated_parts = []
    for i, chunk in enumerate(chunks):
        if progress_callback:
            progress_callback(i, len(chunks))
        translated_parts.append(_translate_chunk(chunk))
    if progress_callback:
        progress_callback(len(chunks), len(chunks))

    return "\n\n".join(translated_parts)

_di_client = None
_di_available = False

def _init_doc_intelligence():
    """Initialize Azure Document Intelligence client."""
    global _di_client, _di_available
    if _di_client is not None:
        return _di_available

    try:
        from azure.ai.documentintelligence import DocumentIntelligenceClient
        from azure.core.credentials import AzureKeyCredential

        key = (os.getenv("AZURE_DOC_INTELLIGENCE_KEY", "")
               or os.getenv("AZURE_DOCUMENT_INTELLIGENCE_KEY", ""))
        endpoint = (os.getenv("AZURE_DOC_INTELLIGENCE_ENDPOINT", "")
                    or os.getenv("AZURE_DOCUMENT_INTELLIGENCE_ENDPOINT", ""))

        if key and endpoint:
            _di_client = DocumentIntelligenceClient(
                endpoint=endpoint,
                credential=AzureKeyCredential(key),
            )
            _di_available = True
            logger.info("Azure Document Intelligence initialized")
        else:
            logger.warning("Azure DI credentials not set")
            _di_available = False
    except ImportError:
        logger.warning("azure-ai-documentintelligence not installed")
        _di_available = False
    except Exception as e:
        logger.warning(f"Azure DI init failed: {e}")
        _di_available = False

    return _di_available


def extract_scanned_pdf_azure(file_bytes: bytes) -> tuple[str, int, list]:
    """
    Extract text and tables from a scanned PDF using Azure Document Intelligence.
    Returns (extracted_text, page_count, tables).
    Each table is a dict: {page, row_count, col_count, headers, rows, raw_markdown}.
    """
    from azure.ai.documentintelligence.models import AnalyzeDocumentRequest

    _init_doc_intelligence()
    if not _di_available or _di_client is None:
        raise RuntimeError(
            "Azure Document Intelligence not configured. "
            "Set AZURE_DOC_INTELLIGENCE_KEY and AZURE_DOC_INTELLIGENCE_ENDPOINT in .env"
        )

    # Call Azure DI with retry
    max_retries = 3
    last_error = None
    for attempt in range(max_retries):
        try:
            poller = _di_client.begin_analyze_document(
                "prebuilt-layout",
                AnalyzeDocumentRequest(bytes_source=file_bytes),
            )
            result = poller.result()
            break
        except Exception as e:
            last_error = e
            if attempt < max_retries - 1:
                delay = 2 * (2 ** attempt)
                logger.warning(f"DI attempt {attempt+1}/{max_retries} failed: {e}. Retrying in {delay}s...")
                time.sleep(delay)
    else:
        raise RuntimeError(f"Azure Document Intelligence failed after {max_retries} attempts: {last_error}")

    # Extract text page by page
    full_text = ""
    page_count = len(result.pages) if result.pages else 0

    if result.pages:
        for page in result.pages:
            page_num = page.page_number
            page_lines = []
            if page.lines:
                for line in page.lines:
                    page_lines.append(line.content)
            if page_lines:
                full_text += f"--- Page {page_num} ---\n" + "\n".join(page_lines) + "\n\n"
    elif result.content:
        full_text = result.content
        page_count = 1

    # Extract tables
    tables = []
    if result.tables:
        for t_idx, table in enumerate(result.tables):
            row_count = table.row_count
            col_count = table.column_count

            # Determine which page this table is on
            table_page = None
            if table.bounding_regions:
                table_page = table.bounding_regions[0].page_number

            # Build a 2D grid from cells
            grid = [["" for _ in range(col_count)] for _ in range(row_count)]
            for cell in table.cells:
                r, c = cell.row_index, cell.column_index
                if r < row_count and c < col_count:
                    grid[r][c] = (cell.content or "").strip()

            # First row as headers, rest as data rows
            headers = grid[0] if grid else []
            rows = grid[1:] if len(grid) > 1 else []

            # Build markdown representation
            md_lines = []
            if headers:
                md_lines.append("| " + " | ".join(headers) + " |")
                md_lines.append("| " + " | ".join(["---"] * len(headers)) + " |")
            for row in rows:
                md_lines.append("| " + " | ".join(row) + " |")
            raw_markdown = "\n".join(md_lines)

            tables.append({
                "index": t_idx + 1,
                "page": table_page,
                "row_count": row_count,
                "col_count": col_count,
                "headers": headers,
                "rows": rows,
                "markdown": raw_markdown,
            })

    return full_text.strip(), page_count, tables


def _clean_ocr_text(text: str) -> str:
    """Clean OCR artifacts from extracted text."""
    text = re.sub(r'\n{3,}', '\n\n', text)
    text = re.sub(r'[ \t]+', ' ', text)
    text = re.sub(r'(\w)-\n(\w)', r'\1\2', text)  # fix hyphenation
    text = re.sub(r'[|]{2,}', '', text)
    text = re.sub(r'_{3,}', '', text)
    return text.strip()
